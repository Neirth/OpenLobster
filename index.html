build from source 
# Requires Go 1.22+ and pnpm
pnpm install
pnpm build --filter=@openlobster/backend

# Dashboard at http://127.0.0.1:8080
./dist/openlobster
docker 
docker run -p 8080:8080 \
  -e OPENLOBSTER_GRAPHQL_HOST=0.0.0.0 \
  -e OPENLOBSTER_GRAPHQL_AUTH_TOKEN=your-secret \
  -v ~/.openlobster/data:/app/data \
  ghcr.io/neirth/openlobster/openlobster:latest
minimal environment variables 
# Agent identity
OPENLOBSTER_AGENT_NAME=my-agent

# Database — SQLite for local use
OPENLOBSTER_DATABASE_DRIVER=sqlite
OPENLOBSTER_DATABASE_DSN=./data/openlobster.db

# Auth — set before exposing to the network
OPENLOBSTER_GRAPHQL_AUTH_TOKEN=strong-secret

# Memory — file backend, no Neo4j required
OPENLOBSTER_MEMORY_BACKEND=file
OPENLOBSTER_MEMORY_FILE_PATH=./data/memory

# AI provider — pick one
OPENLOBSTER_PROVIDERS_OLLAMA_ENDPOINT=http://localhost:11434
# or
OPENLOBSTER_PROVIDERS_ANTHROPIC_API_KEY=sk-ant-…
# or
OPENLOBSTER_PROVIDERS_OPENAI_API_KEY=sk-…
